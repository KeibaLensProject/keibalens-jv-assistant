```python
@dataclass
class ODDS_WIDE_INFO:
    """ワイドオッズ"""
    Kumi: str       # 組番
    OddsLow: str    # 最低オッズ
    OddsHigh: str   # 最高オッズ
    Ninki: str      # 人気順

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 4),
            MidB2S(b, 5, 5),
            MidB2S(b, 10, 5),
            MidB2S(b, 15, 3),
        )

@dataclass
class JV_O3_ODDS_WIDE:
    head: RECORD_ID                             # <レコードヘッダー>
    id: RACE_ID                                 # <競走識別情報>
    HappyoTime: MDHM                            # 発表月日時分
    TorokuTosu: str                             # 登録頭数
    SyussoTosu: str                             # 出走頭数
    WideFlag: str                               # 発売フラグ ワイド
    OddsWideInfo: List[ODDS_WIDE_INFO]          # <ワイド票数オッズ>
    TotalHyosuWide: str                         # ワイド票数合計
    crlf: str                                   # レコード区切り

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RECORD_ID.SetDataB(MidB2B(b, 1, 11)),
            RACE_ID.SetDataB(MidB2B(b, 12, 16)),
            MDHM.SetDataB(MidB2B(b, 28, 8)),
            MidB2S(b, 36, 2),
            MidB2S(b, 38, 2),
            MidB2S(b, 40, 1),
            [ODDS_WIDE_INFO.SetDataB(MidB2B(b, 41 + 17 * i, 17)) for i in range(153)],
            MidB2S(b, 2642, 11),
            MidB2S(b, 2653, 2),
        )
```
