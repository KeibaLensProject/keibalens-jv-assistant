class HATUKIJYO_INFO:
    """初騎乗情報"""
    Hatukijyoid: RACE_ID    # 年月日場回日R
    SyussoTosu: str         # 出走頭数
    KettoNum: str           # 血統登録番号
    Bamei: str              # 馬名
    KakuteiJyuni: str       # 確定着順
    IJyoCD: str             # 異常区分コード

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RACE_ID.SetDataB(MidB2B(b, 1, 16)),
            MidB2S(b, 17, 2),
            MidB2S(b, 19, 10),
            MidB2S(b, 29, 36),
            MidB2S(b, 65, 2),
            MidB2S(b, 67, 1),
        )
