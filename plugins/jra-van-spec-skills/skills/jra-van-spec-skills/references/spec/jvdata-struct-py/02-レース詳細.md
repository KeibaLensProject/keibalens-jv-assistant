@dataclass
class CORNER_INFO:
    """コーナー通過順位"""
    Corner: str     # コーナー
    Syukaisu: str   # 周回数
    Jyuni: str      # 各通過順位

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 1),
            MidB2S(b, 2, 1),
            MidB2S(b, 3, 70),
        )

    KyoriBefore: str                # 変更前距離
    TrackCD: str                    # トラックコード

    CourseKubunCD: str              # コース区分
    CourseKubunCDBefore: str        # 変更前コース区分

    HonsyokinBefore: List[str]      # 変更前本賞金
    Fukasyokin: List[str]           # 付加賞金
    FukasyokinBefore: List[str]     # 変更前付加賞金
    HassoTime: str                  # 発走時刻
    HassoTimeBefore: str            # 変更前発走時刻
    TorokuTosu: str                 # 登録頭数
    SyussoTosu: str                 # 出走頭数
    NyusenTosu: str                 # 入線頭数
    TenkoBaba: TENKO_BABA_INFO      # 天候・馬場状態コード
    LapTime: List[str]              # ラップタイム
    SyogaiMileTime: str             # 障害マイルタイム
    HaronTimeS3: str                # 前３ハロンタイム
    HaronTimeS4: str                # 前４ハロンタイム
    HaronTimeL3: str                # 後３ハロンタイム
    HaronTimeL4: str                # 後４ハロンタイム
    CornerInfo: List[CORNER_INFO]   # <コーナー通過順位>
    RecordUpKubun: str              # レコード更新区分
    crlf: str                       # レコード区切り

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RECORD_ID.SetDataB(MidB2B(b, 1, 11)),
            RACE_ID.SetDataB(MidB2B(b, 12, 16)),
            RACE_INFO.SetDataB(MidB2B(b, 28, 587)),
            MidB2S(b, 615, 1),
            MidB2S(b, 616, 1),
            RACE_JYOKEN.SetDataB(MidB2B(b, 617, 21)),
            MidB2S(b, 638, 60),
            MidB2S(b, 698, 4),
            MidB2S(b, 702, 4),
            MidB2S(b, 706, 2),
            MidB2S(b, 708, 2),
            MidB2S(b, 710, 2),
            MidB2S(b, 712, 2),
            [MidB2S(b, 714 + 8 * i, 8) for i in range(7)],
            [MidB2S(b, 770 + 8 * i, 8) for i in range(5)],
            [MidB2S(b, 810 + 8 * i, 8) for i in range(5)],
            [MidB2S(b, 850 + 8 * i, 8) for i in range(3)],
            MidB2S(b, 874, 4),
            MidB2S(b, 878, 4),
            MidB2S(b, 882, 2),
            MidB2S(b, 884, 2),
            MidB2S(b, 886, 2),
            TENKO_BABA_INFO.SetDataB(MidB2B(b, 888, 3)),

            MidB2S(b, 973, 3),
            MidB2S(b, 976, 3),
            MidB2S(b, 979, 3),
            [CORNER_INFO.SetDataB(MidB2B(b, 982 + 72 * i, 72)) for i in range(4)],
            MidB2S(b, 1270, 1),
            MidB2S(b, 1271, 2),
        )
