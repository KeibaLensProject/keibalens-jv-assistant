@dataclass
class ODDS_TANSYO_INFO:
    """単勝オッズ"""
    Umaban: str     # 馬番
    Odds: str       # オッズ
    Ninki: str      # 人気順

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 2),
            MidB2S(b, 3, 4),
            MidB2S(b, 7, 2),
        )

@dataclass
class ODDS_FUKUSYO_INFO:
    """複勝オッズ"""
    Umaban: str     # 馬番
    OddsLow: str    # 最低オッズ
    OddsHigh: str   # 最高オッズ
    Ninki: str      # 人気順

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 2),
            MidB2S(b, 3, 4),
            MidB2S(b, 7, 4),
            MidB2S(b, 11, 2),
        )

@dataclass
class ODDS_WAKUREN_INFO:
    """枠連オッズ"""
    Kumi: str       # 組番
    Odds: str       # オッズ
    Ninki: str      # 人気順

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 2),
            MidB2S(b, 3, 5),
            MidB2S(b, 8, 2),
        )

@dataclass
class JV_O1_ODDS_TANFUKUWAKU:
    head: RECORD_ID                             # <レコードヘッダー>
    id: RACE_ID                                 # <競走識別情報>
    HappyoTime: MDHM                            # 発表月日時分
    TorokuTosu: str                             # 登録頭数
    SyussoTosu: str                             # 出走頭数
    TansyoFlag: str                             # 発売フラグ 単勝
    FukusyoFlag: str                            # 発売フラグ 複勝
    WakurenFlag: str                            # 発売フラグ 枠連
    FukuChakuBaraiKey: str                      # 複勝着払キー
    OddsTansyoInfo: List[ODDS_TANSYO_INFO]      # <単勝オッズ>
    OddsFukusyoInfo: List[ODDS_FUKUSYO_INFO]    # <複勝票数オッズ>
    OddsWakurenInfo: List[ODDS_WAKUREN_INFO]    # <枠連票数オッズ>
    TotalHyosuTansyo: str                       # 単勝票数合計
    TotalHyosuFukusyo: str                      # 複勝票数合計
    TotalHyosuWakuren: str                      # 枠連票数合計
    crlf: str                                   # レコード区切り

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RECORD_ID.SetDataB(MidB2B(b, 1, 11)),
            RACE_ID.SetDataB(MidB2B(b, 12, 16)),
            MDHM.SetDataB(MidB2B(b, 28, 8)),
            MidB2S(b, 36, 2),
            MidB2S(b, 38, 2),
            MidB2S(b, 40, 1),
            MidB2S(b, 41, 1),
            MidB2S(b, 42, 1),
            MidB2S(b, 43, 1),
            [ODDS_TANSYO_INFO.SetDataB(MidB2B(b, 44 + 8 * i, 8)) for i in range(28)],
            [ODDS_FUKUSYO_INFO.SetDataB(MidB2B(b, 268 + 12 * i, 12)) for i in range(28)],
            [ODDS_WAKUREN_INFO.SetDataB(MidB2B(b, 604 + 9 * i, 9)) for i in range(36)],
            MidB2S(b, 928, 11),
            MidB2S(b, 939, 11),
            MidB2S(b, 950, 11),
            MidB2S(b, 961, 2),
        )
