# 1.特別登録馬 (Python)

出典: `context/JVData_Struct.py` (JRA-VAN提供サンプル, 最終更新 2026-07-31, SDK Ver.5.0.0)

| クラス (`@dataclass`) | 概要 |
|---|---|
| `TOKUUMA_INFO` | 登録馬毎情報 |
| `JV_TK_TOKUUMA` | 特別登録馬 |

```python
@dataclass
class TOKUUMA_INFO:
    """登録馬毎情報"""
    Num: str                # 連番
    KettoNum: str           # 血統登録番号
    Bamei: str              # 馬名
    UmaKigoCD: str          # 馬記号コード
    SexCD: str              # 性別コード
    TozaiCD: str            # 調教師東西所属コード
    ChokyosiCode: str       # 調教師コード
    ChokyosiRyakusyo: str   # 調教師名略称
    Futan: str              # 負担重量
    Koryu: str              # 交流区分

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 3),
            MidB2S(b, 4, 10),
            MidB2S(b, 14, 36),
            MidB2S(b, 50, 2),
            MidB2S(b, 52, 1),
            MidB2S(b, 53, 1),
            MidB2S(b, 54, 5),
            MidB2S(b, 59, 8),
            MidB2S(b, 67, 3),
            MidB2S(b, 70, 1),
        )

@dataclass
class JV_TK_TOKUUMA:
    head: RECORD_ID                     # <レコードヘッダー>
    id: RACE_ID                         # <競走識別情報>
    RaceInfo: RACE_INFO                 # <レース情報>
    GradeCD: str                        # グレードコード
    JyokenInfo: RACE_JYOKEN             # <競走条件コード>
    Kyori: str                          # 距離
    TrackCD: str                        # トラックコード
    CourseKubunCD: str                  # コース区分
    HandiDate: YMD                      # ハンデ発表日
    TorokuTosu: str                     # 登録頭数
    TokuUmaInfo: List[TOKUUMA_INFO]     # <登録馬毎情報>
    crlf: str                           # レコード区切

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RECORD_ID.SetDataB(MidB2B(b, 1, 11)),
            RACE_ID.SetDataB(MidB2B(b, 12, 16)),
            RACE_INFO.SetDataB(MidB2B(b, 28, 587)),
            MidB2S(b, 615, 1),
            RACE_JYOKEN.SetDataB(MidB2B(b, 616, 21)),
            MidB2S(b, 637, 4),
            MidB2S(b, 641, 2),
            MidB2S(b, 643, 2),
            YMD.SetDataB(MidB2B(b, 645, 8)),
            MidB2S(b, 653, 3),
            [TOKUUMA_INFO.SetDataB(MidB2B(b, 656 + 70 * i, 70)) for i in range(300)],
            MidB2S(b, 21656, 2),
        )
```
