# 30.重勝式(WIN5) (Python)

出典: `context/JVData_Struct.py` (JRA-VAN提供サンプル, 最終更新 2026-07-31, SDK Ver.5.0.0)

| クラス (`@dataclass`) | 概要 |
|---|---|
| `WF_RACE_INFO` | 重勝式対象レース情報 |
| `WF_YUKO_HYO_INFO` | 有効票数情報 |
| `WF_PAY_INFO` | 重勝式払戻情報 |
| `JV_WF_INFO` | 重勝式(WIN5) |

```python
@dataclass
class WF_RACE_INFO:
    """重勝式対象レース情報"""
    JyoCD: str      # 競馬場コード
    Kaiji: str      # 開催回[第N回]
    Nichiji: str    # 開催日目[N日目]
    RaceNum: str    # レース番号

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 2),
            MidB2S(b, 3, 2),
            MidB2S(b, 5, 2),
            MidB2S(b, 7, 2),
        )

@dataclass
class WF_YUKO_HYO_INFO:
    """有効票数情報"""
    Yuko_Hyo: str   # 有効票数

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 11),
        )

@dataclass
class WF_PAY_INFO:
    """重勝式払戻情報"""
    Kumiban: str        # 組番
    Pay: str            # 重勝式払戻金
    Tekichu_Hyo: str    # 的中票数

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            MidB2S(b, 1, 10),
            MidB2S(b, 11, 9),
            MidB2S(b, 20, 10),
        )

@dataclass
class JV_WF_INFO:
    head: RECORD_ID                         # <レコードヘッダー>
    KaisaiDate: YMD                         # 開催年月日
    reserved1: str                          # 予備
    WFRaceInfo: List[WF_RACE_INFO]          # <重勝式対象レース情報>
    reserved2: str                          # 予備
    Hatsubai_Hyo: str                       # 重勝式発売票数
    WFYukoHyoInfo: List[WF_YUKO_HYO_INFO]   # <有効票数情報>
    HenkanFlag: str                         # 返還フラグ
    FuseiritsuFlag: str                     # 不成立フラグ
    TekichunashiFlag: str                   # 的中無フラグ
    COShoki: str                            # キャリーオーバー金額初期
    COZanDaka: str                          # キャリーオーバー金額残高
    WFPayInfo: List[WF_PAY_INFO]            # <重勝式払戻情報>
    crlf: str                               # レコード区切り

    # データセット
    @classmethod
    def SetDataB(cls, b):
        return cls(
            RECORD_ID.SetDataB(MidB2B(b, 1, 11)),
            YMD.SetDataB(MidB2B(b, 12, 8)),
            MidB2S(b, 20, 2),
            [WF_RACE_INFO.SetDataB(MidB2B(b, 22 + 8 * i, 8)) for i in range(5)],
            MidB2S(b, 62, 6),
            MidB2S(b, 68, 11),
            [WF_YUKO_HYO_INFO.SetDataB(MidB2B(b, 79 + 11 * i, 11)) for i in range(5)],
            MidB2S(b, 134, 1),
            MidB2S(b, 135, 1),
            MidB2S(b, 136, 1),
            MidB2S(b, 137, 15),
            MidB2S(b, 152, 15),
            [WF_PAY_INFO.SetDataB(MidB2B(b, 167 + 29 * i, 29)) for i in range(243)],
            MidB2S(b, 7214, 2),
        )
```
